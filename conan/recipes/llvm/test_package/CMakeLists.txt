cmake_minimum_required(VERSION 3.31)
project(llvm_package_probe LANGUAGES C CXX)

find_package(LLVM 23.1.2 EXACT CONFIG REQUIRED)
if(NOT LLVM_PACKAGE_VERSION VERSION_EQUAL "23.1.2" OR NOT TARGET LLVMSupport)
  message(FATAL_ERROR "Exact native LLVM 23.1.2 config/LLVMSupport required")
endif()
if(NOT LLVM_BUILD_TYPE STREQUAL "Release" OR
   NOT LLVM_ABI_BREAKING_CHECKS STREQUAL "FORCE_OFF")
  message(FATAL_ERROR "Expected optimized LLVM with ABI-breaking checks disabled")
endif()
if(LLVM_ENABLE_SHARED_LIBS OR NOT LLVM_TARGETS_TO_BUILD STREQUAL "X86")
  message(FATAL_ERROR "Expected static X86-only LLVM")
endif()
get_target_property(_support_type LLVMSupport TYPE)
if(NOT _support_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "LLVMSupport must be the upstream static imported target")
endif()

if(WIN32)
  if(NOT DEFINED VEYR_EXPECTED_MSVC_RUNTIME)
    message(FATAL_ERROR "Selected Windows runtime identity is required")
  endif()
  # Evaluate identity at the package's Release configuration, not consumer Debug.
  set(_llvm_runtime "${CMAKE_MSVC_RUNTIME_LIBRARY}")
  string(FIND "${_llvm_runtime}"
    "$<$<CONFIG:${LLVM_BUILD_TYPE}>:${VEYR_EXPECTED_MSVC_RUNTIME}>" _runtime_match)
  if(NOT _llvm_runtime STREQUAL VEYR_EXPECTED_MSVC_RUNTIME AND _runtime_match EQUAL -1)
    message(FATAL_ERROR "LLVM runtime '${_llvm_runtime}' differs from ${VEYR_EXPECTED_MSVC_RUNTIME}")
  endif()
  # LLVMConfig sets the global variable; restore selected consumer CRT deliberately.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "${VEYR_EXPECTED_MSVC_RUNTIME}")
endif()

# Map only native LLVM targets; never alter configuration selection for other packages.
foreach(_target IN LISTS LLVM_AVAILABLE_LIBS)
  if(TARGET "${_target}")
    get_target_property(_imported "${_target}" IMPORTED)
    if(_imported)
      set_property(TARGET "${_target}" PROPERTY MAP_IMPORTED_CONFIG_DEBUG Release)
    endif()
  endif()
endforeach()

add_executable(llvm_package_probe src/llvm_probe.cpp)
target_compile_features(llvm_package_probe PRIVATE cxx_std_23)
target_include_directories(llvm_package_probe SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(_llvm_definitions NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(llvm_package_probe PRIVATE ${_llvm_definitions})
target_link_libraries(llvm_package_probe PRIVATE LLVMSupport)
if(WIN32)
  set_property(TARGET llvm_package_probe PROPERTY MSVC_RUNTIME_LIBRARY "${VEYR_EXPECTED_MSVC_RUNTIME}")
endif()
message(STATUS "LLVM probe: ${LLVM_PACKAGE_VERSION}; ${LLVM_BUILD_TYPE}; targets=${LLVM_TARGETS_TO_BUILD}; consumer=${CMAKE_BUILD_TYPE}")
